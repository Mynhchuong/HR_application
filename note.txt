CREATE TABLE HRMS.HR_MEAL_LOCK (
    ID              NUMBER          PRIMARY KEY,
    LOCK_DATE       DATE            NOT NULL,        -- ngày áp luật (TRUNC = 00:00)
    LOCK_LUNCH      CHAR(1) DEFAULT 'N' NOT NULL,    -- Y = khoá đổi món trưa
    LOCK_OT         CHAR(1) DEFAULT 'N' NOT NULL,    -- Y = khoá đổi món tăng ca
    CUTOFF_DT       DATE,                            -- thời điểm chốt; NULL = khoá toàn bộ
    NOTE            VARCHAR2(500),
    IS_ACTIVE       CHAR(1) DEFAULT 'Y' NOT NULL,
    INST_ID         VARCHAR2(11),
    INST_DT         DATE DEFAULT SYSDATE,
    UPDT_ID         VARCHAR2(11),
    UPDT_DT         DATE
);

CREATE SEQUENCE HRMS.SEQ_HR_MEAL_LOCK START WITH 1 INCREMENT BY 1 NOCACHE;

CREATE OR REPLACE TRIGGER HRMS.TRG_HR_MEAL_LOCK_BI
BEFORE INSERT ON HRMS.HR_MEAL_LOCK
FOR EACH ROW
BEGIN
    IF :NEW.ID IS NULL THEN
        SELECT HRMS.SEQ_HR_MEAL_LOCK.NEXTVAL INTO :NEW.ID FROM DUAL;
    END IF;
END;
/

CREATE INDEX HRMS.IDX_HR_MEAL_LOCK_DATE
    ON HRMS.HR_MEAL_LOCK(LOCK_DATE, IS_ACTIVE);
 ALTER TABLE HRMS.HR_MEAL_LOCK ADD (
       LOCK_MAN  CHAR(1) DEFAULT 'N' NOT NULL,
      LOCK_NHE  CHAR(1) DEFAULT 'N' NOT NULL,
      LOCK_CHAY CHAR(1) DEFAULT 'N' NOT NULL
  );
  ALTER TABLE HRMS.HR_NOTIFICATIONS MODIFY LINK_ACTION VARCHAR2(500);
