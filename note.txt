-- 17. HR_TRAINING_AUDIT — log bulk action (assign, approve, drop, publish, ...)
-- Dùng để tra cứu "ai bulk assign 200 người vào Class X lúc nào"
-- ============================================================
CREATE TABLE HRMS.HR_TRAINING_AUDIT (
    ID              NUMBER          NOT NULL,
    ACTION          VARCHAR2(40)    NOT NULL,       -- BULK_ASSIGN | BULK_APPROVE | BULK_REJECT | BULK_DROP | PUBLISH | UNPUBLISH | CANCEL_CLASS | RESCHEDULE_SESSION
    CLASS_ID        NUMBER,
    SESSION_ID      NUMBER,                          -- nullable
    TEST_ID         NUMBER,                          -- nullable
    AFFECTED_COUNT  NUMBER          DEFAULT 0 NOT NULL,
    PAYLOAD         NVARCHAR2(2000),                 -- JSON tùy chọn: {"empcds":["...","..."], "reason":"..."}
    ACTOR_EMPCD     VARCHAR2(20),
    INST_DT         DATE            DEFAULT SYSDATE,

    CONSTRAINT PK_HR_TR_AUDIT   PRIMARY KEY (ID)
) TABLESPACE HRMSD;

CREATE SEQUENCE HRMS.SEQ_HR_TR_AUDIT START WITH 1 INCREMENT BY 1 NOCACHE;

CREATE OR REPLACE TRIGGER HRMS.TRG_HR_TR_AUDIT_BI
BEFORE INSERT ON HRMS.HR_TRAINING_AUDIT
FOR EACH ROW
BEGIN
    IF :NEW.ID IS NULL THEN
        SELECT HRMS.SEQ_HR_TR_AUDIT.NEXTVAL INTO :NEW.ID FROM DUAL;
    END IF;
    IF :NEW.INST_DT IS NULL THEN :NEW.INST_DT := SYSDATE; END IF;
END;
/
ALTER TRIGGER HRMS.TRG_HR_TR_AUDIT_BI ENABLE;

-- ============================================================
-- 18. HR_TRAINING_NOTI_QUEUE — persistent queue + retry cho FCM
-- Notification thất bại (FCM down, token invalid) → save vào đây, batch job retry.
-- ============================================================
CREATE TABLE HRMS.HR_TRAINING_NOTI_QUEUE (
    ID              NUMBER          NOT NULL,
    TEMPLATE_KEY    VARCHAR2(60)    NOT NULL,        -- TRAINING_ASSIGNED, TRAINING_SESSION_TODAY, ...
    TARGET_EMPCD    VARCHAR2(20)    NOT NULL,
    PLACEHOLDERS    NVARCHAR2(2000),                 -- JSON: {"className":"...","startDate":"..."}
    RELATED_CLASS_ID  NUMBER,
    RELATED_SESSION_ID NUMBER,
    RELATED_TEST_ID   NUMBER,

    -- CLAIMED status cho UPDATE-claim pattern (§13 rules — Oracle 10 không có FOR UPDATE SKIP LOCKED)
    STATUS          VARCHAR2(15)    DEFAULT 'PENDING' NOT NULL,
                                                     -- PENDING | CLAIMED | SENT | FAILED
    CLAIMED_BY      VARCHAR2(50),                    -- worker instance ID
    CLAIMED_DT      DATE,                            -- stuck > 5 min → reset về PENDING

    RETRY_COUNT     NUMBER          DEFAULT 0 NOT NULL,
    LAST_ERROR      NVARCHAR2(500),
    NEXT_ATTEMPT_DT DATE            DEFAULT SYSDATE NOT NULL,

    INST_DT         DATE            DEFAULT SYSDATE,
    SENT_DT         DATE,

    CONSTRAINT PK_HR_TR_NOTIQ   PRIMARY KEY (ID),
    CONSTRAINT CHK_TRNQ_STATUS  CHECK (STATUS IN ('PENDING','CLAIMED','SENT','FAILED'))
) TABLESPACE HRMSD;

CREATE SEQUENCE HRMS.SEQ_HR_TR_NOTIQ START WITH 1 INCREMENT BY 1 NOCACHE;

CREATE OR REPLACE TRIGGER HRMS.TRG_HR_TR_NOTIQ_BI
BEFORE INSERT ON HRMS.HR_TRAINING_NOTI_QUEUE
FOR EACH ROW
BEGIN
    IF :NEW.ID IS NULL THEN
        SELECT HRMS.SEQ_HR_TR_NOTIQ.NEXTVAL INTO :NEW.ID FROM DUAL;
    END IF;
    IF :NEW.INST_DT IS NULL THEN :NEW.INST_DT := SYSDATE; END IF;
END;
/
ALTER TRIGGER HRMS.TRG_HR_TR_NOTIQ_BI ENABLE;

CREATE INDEX HRMS.IDX_TRNQ_PENDING ON HRMS.HR_TRAINING_NOTI_QUEUE (STATUS, NEXT_ATTEMPT_DT) TABLESPACE HRMSD;


COMMIT;
